# Build the image (first run and after changes to Dockerfile/requirements.txt)
docker compose build

# Build without cache (if something gets stuck)
docker compose build --no-cache

# Default software mode (checks only, no USB or RF)
docker compose up

# Run software mode in the background
docker compose up -d

# Hardware mode (hardware profile, USB passthrough)
docker compose --profile hardware up gps-simulator-hardware

# Stop containers (press Ctrl+C if running in foreground)
docker compose down

# Stop and remove containers and networks
docker compose down --remove-orphans

# Remove the project image (if a clean rebuild is required)
docker compose down --rmi local

# Validate the Compose configuration (without starting containers)
docker compose config

# View logs from running containers
docker compose logs

# Follow logs in real time
docker compose logs -f

# List running containers for the project
docker compose ps

# List all project images
docker compose images

1. Checks only (no generation, no RF)

docker compose run --rm gps-simulator
docker compose run --rm gps-simulator --mode software

2. Checks + IQ generation

# Generate gpssim.bin and exit (without monitoring)
docker compose run --rm gps-simulator --mode software --generate --detached

# Generate IQ data and start monitoring (no RF; press Ctrl+C to stop)
docker compose run --rm gps-simulator --mode software --generate

3. Hardware checks (HackRF accessible from the container)

# Check whether the container can detect HackRF (no RF transmission)
docker compose --profile hardware run --rm gps-simulator-hardware --mode hardware --detached

4. Signal transmission via HackRF (without ZED-F9P)

# Generate IQ data and transmit continuously (press Ctrl+C to stop)
docker compose --profile hardware run --rm gps-simulator-hardware --mode hardware --generate --tx

# Transmit an existing gpssim.bin file
docker compose --profile hardware run --rm gps-simulator-hardware --mode hardware --tx

5. RF test with ZED-F9P

# Generate IQ data, transmit, and read UBX data from the receiver (90 seconds)
docker compose --profile hardware run --rm gps-simulator-hardware --mode hardware --generate --rf

# RF test without generation (uses the existing gpssim.bin file)
docker compose --profile hardware run --rm gps-simulator-hardware --mode hardware --rf

6. Full run (same as start-gps-sim.bat)

docker compose --profile hardware run --rm gps-simulator-hardware --mode hardware --generate --tx
